Run the worker through stock DuckDB 2.0.0 with VGI from query_farm, add repeatable tests, and leave vgi-worker.json with the actual launch argv and catalog demo in this project. cd . && mkdir -p src test && cat > src/slugify.ts <<'EOF'
// Pure slugify logic, independent of VGI so it can be unit tested directly.

// Latin letters that Unicode NFKD does not decompose into ASCII + combining marks.
const SPECIAL: Record<string, string> = {
  "ß": "ss", "ẞ": "ss",
  "æ": "ae", "Æ": "ae",
  "œ": "oe", "Œ": "oe",
  "ø": "o", "Ø": "o",
  "đ": "d", "Đ": "d",
  "ð": "d", "Ð": "d",
  "þ": "th", "Þ": "th",
  "ł": "l", "Ł": "l",
  "ħ": "h", "Ħ": "h",
  "ı": "i",
  "ŀ": "l", "Ŀ": "l",
  "ŧ": "t", "Ŧ": "t",
  "ŋ": "n", "Ŋ": "n",
  "ĸ": "k",
};
const SPECIAL_RE = new RegExp(`[${Object.keys(SPECIAL).join("")}]`, "gu");

/**
 * Lowercase, transliterate accented Latin characters to ASCII, collapse runs of
 * punctuation/whitespace/symbols into a single hyphen, and trim edge hyphens.
 * Letters and digits from non-Latin scripts are kept as-is.
 */
export function slugify(input: string): string {
  return input
    .replace(SPECIAL_RE, (ch) => SPECIAL[ch]!)
    .normalize("NFKD")
    .replace(/\p{M}+/gu, "")
    .toLowerCase()
    .replace(/[^\p{L}\p{N}]+/gu, "-")
    .replace(/^-+|-+$/g, "");
}
EOF
cat > worker.ts <<'EOF'
// VGI worker exposing demo.main.slugify(VARCHAR) -> VARCHAR over stdio.
// stdout carries Arrow protocol bytes; never log to stdout.
import { Worker, defineScalarFunction, str, FunctionStability } from "@query-farm/vgi";
import { slugify } from "./src/slugify.ts";

export const slugifyFn = defineScalarFunction({
  name: "slugify",
  description:
    "Convert text to a URL slug: lowercase, transliterate accented Latin to ASCII, " +
    "collapse punctuation/whitespace runs to '-', trim edge hyphens. NULL in, NULL out.",
  params: { text: str },
  argDocs: { text: "Text to slugify" },
  returns: str,
  stability: FunctionStability.CONSISTENT,
  examples: [],
  compute: (batch) =>
    Array.from(batch.getChildAt(0)!, (value) =>
      value == null ? null : slugify(value as string)),
});

new Worker({
  catalog: {
    name: "demo",
    description: "Demo catalog with text utilities",
    schemas: [{ path: ["main"], functions: [slugifyFn] }],
  },
}).run();
EOF

cat > test/slugify.test.ts <<'EOF'
import { test } from "node:test";
import assert from "node:assert/strict";
import { slugify } from "../src/slugify.ts";

const cases: [string, string][] = [
  ["Hello, World!", "hello-world"],
  ["  --Crème Brûlée--  ", "creme-brulee"],
  ["Ångström Øresund Straße", "angstrom-oresund-strasse"],
  ["Łódź & Kraków", "lodz-krakow"],
  ["Æsir/Œuvre__Þorn", "aesir-oeuvre-thorn"],
  ["multiple   spaces\t\nand...dots", "multiple-spaces-and-dots"],
  ["already-a-slug", "already-a-slug"],
  ["Version 2.0 (beta)", "version-2-0-beta"],
  ["İstanbul", "istanbul"],
  ["!!!", ""],
  ["", ""],
];

for (const [input, expected] of cases) {
  test(`slugify(${JSON.stringify(input)})`, () => {
    assert.equal(slugify(input), expected);
    assert.match(slugify(input), /^(?:[a-z0-9]+(?:-[a-z0-9]+)*)?$/);
  });
}
EOF
